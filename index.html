1. Breadth-First Search (BFS) and Depth-First Search (DFS)

Python 
graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F'],
    'D': [],
    'E': ['G'],
    'F': [],
    'G': []
}

def bfs(graph, start):
    visited = []
    queue = [start]
    while queue:
        node = queue.pop(0) 
        if node not in visited:
            visited.append(node)
            queue.extend(graph[node])
    return visited

def dfs(graph, start):
    visited = []
    stack = [start]
    while stack:
        node = stack.pop() 
        if node not in visited:
            visited.append(node)
            stack.extend(reversed(graph[node]))
    return visited

start_node = 'A'
print("Breadth-First Search (BFS):")
print(bfs(graph, start_node))
print("\nDepth-First Search (DFS):")
print(dfs(graph, start_node))

2. 8-Puzzle Problem using State-Space Representation

Python 
from collections import deque

goal_state = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 0]
]

def to_tuple(state):
    return tuple(tuple(row) for row in state)

def find_blank(state):
    for i in range(3):
        for j in range(3):
            if state[i][j] == 0:
                return i, j

def get_neighbors(state):
    neighbors = []
    x, y = find_blank(state)
    moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]  
    for dx, dy in moves:
        new_x = x + dx
        new_y = y + dy
        if 0 <= new_x < 3 and 0 <= new_y < 3:
            new_state = [row[:] for row in state]
            new_state[x][y], new_state[new_x][new_y] = \
                new_state[new_x][new_y], new_state[x][y]
            neighbors.append(new_state)
    return neighbors

def solve_puzzle(start_state):
    queue = deque()
    visited = set()
    queue.append((start_state, []))
    visited.add(to_tuple(start_state))
    while queue:
        current_state, path = queue.popleft()

        if current_state == goal_state:
            return path + [current_state]
        for neighbor in get_neighbors(current_state):
            if to_tuple(neighbor) not in visited:
                visited.add(to_tuple(neighbor))
                queue.append((neighbor, path + [current_state]))
    return None

def print_state(state):
    for row in state:
        print(row)
    print()

start_state = [
    [1, 0, 3],
    [4, 2, 6],
    [7, 5, 8]
]

solution = solve_puzzle(start_state)
if solution:
    print("Solution Found")
    print("Number of moves:", len(solution) - 1)
    for step, state in enumerate(solution):
        print("\nStep", step)
        print_state(state)
else:
    print("No solution found.")

3. A* Search Algorithm

Python 
import heapq

graph = {
    'A': [('B', 2), ('C', 4)],
    'B': [('D', 7), ('E', 3)],
    'C': [('F', 2)],
    'D': [('G', 1)],
    'E': [('G', 5)],
    'F': [('G', 3)],
    'G': []
}

heuristic = {
    'A': 9,
    'B': 8,
    'C': 5,
    'D': 1,
    'E': 5,
    'F': 3,
    'G': 0
}

def astar(start, goal):
    queue = []
    heapq.heappush(queue, (heuristic[start], 0, start, [start]))
    visited = []
    
    while queue:
        f, g, current, path = heapq.heappop(queue)
        
        if current in visited:
            continue
            
        print("Current Node :", current)
        print("g(n) =", g)
        print("h(n) =", heuristic[current])
        print("f(n) =", f)
        print()
        
        visited.append(current)
        
        if current == goal:
            return path, g
            
        for neighbor, cost in graph[current]:
            if neighbor not in visited:
                new_g = g + cost
                new_f = new_g + heuristic[neighbor]
                heapq.heappush(queue, (new_f, new_g, neighbor, path + [neighbor]))
                
    return None, None

start = 'A'
goal = 'G'
path, cost = astar(start, goal)

print("-------------------------")
if path:
    print("Optimal Path :", " -> ".join(path))
    print("Total Cost   :", cost)
else:
    print("No path found.")

4. Hill Climbing Algorithm

Python 
# Objective Function 
def objective_function(x): 
    return -x * x + 10 * x 

# Hill Climbing Function 
def hill_climbing(start): 
    current = start 
    step_size = 1 
    
    while True: 
        # Generate neighboring solutions 
        left = current - step_size 
        right = current + step_size 
        
        # Evaluate current and neighboring states 
        current_value = objective_function(current) 
        left_value = objective_function(left) 
        right_value = objective_function(right) 
        
        # Display current state 
        print("Current x =", current, " Objective Value =", current_value) 
        
        # Move to the better neighboring state 
        if left_value > current_value: 
            current = left 
        elif right_value > current_value: 
            current = right 
        # Stop if no better neighbor exists 
        else: 
            return current, current_value 

# Main Program 
start_point = 0 
best_x, best_value = hill_climbing(start_point) 

print("\nBest Solution Found") 
print("Optimal x =", best_x) 
print("Maximum Value =", best_value)

print("Total Testing Images  :", len(X_test))

for i in range(10):
    plt.subplot(2, 5, i + 1)
    plt.imshow(X_train[i], cmap="gray")
    plt.title("Digit: " + str(y_train[i]))
    plt.axis("off")

plt.suptitle("Sample MNIST Handwritten Digits")
plt.tight_layout()
plt.show()

# Use Smaller Dataset for Faster Execution
X_train = X_train[:2000]
y_train = y_train[:2000]
X_test = X_test[:500]
y_test = y_test[:500]

print("\n===== DATA USED =====")
print("Training Images :", len(X_train))
print("Testing Images  :", len(X_test))

X_train = X_train / 255.0
X_test = X_test / 255.0

    tf.keras.layers.Input(shape=(28, 28, 1)),

app.post("/register", (req, res) => {
  if (!req.session.loggedIn) {
    return res.redirect("/");
  }
  const {
    name,
    regno,
    email,
    mobile,
    gender,
    dob,
    department,
    course,
    year,
    address,
    skills,
  } = req.body;

  // Validation
  if (
    !name ||
    !regno ||
    !email ||
    !mobile ||
    !gender ||
    !dob ||
    !department ||
    !course ||
    !year ||
    !address
  ) {
    return res.send(`
      <h2>All required fields must be filled.</h2>
      <a href="/register">Go Back</a>
    `);
  }

  // Email validation regex
  const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  if (!emailPattern.test(email)) {
    return res.send(`
      <h2>Invalid email address.</h2>
      <a href="/register">Go Back</a>
    `);
  }

  // Mobile validation regex
  const mobilePattern = /^[0-9]{10}$/;
  if (!mobilePattern.test(mobile)) {
    return res.send(`
      <h2>Mobile number must contain exactly 10 digits.</h2>
      <a href="/register">Go Back</a>
    `);
  }

  // Store student details in session
  req.session.student = {
    name,
    regno,
    email,
    mobile,
    gender,
    dob,
    department,
    course,
    year,
    address,
    skills: skills || "None",
  };
  res.redirect("/student");
});

// Display student details page

const mongoose = require("mongoose");

const employeeSchema = new mongoose.Schema({
  empId:       { type: String, required: true, unique: true },
  name:        { type: String, required: true },
  email:       { type: String, required: true },
  phone:       { type: String, required: true },
  department:  { type: String, required: true },
  designation: { type: String, required: true },
  salary:      { type: Number, required: true },
  joiningDate: { type: Date, required: true },
  address:     { type: String },
  skills:      { type: [String] }, // Array of strings
});

module.exports = mongoose.model("Employee", employeeSchema);

File 3: server.js (Clean Indentation & No Stray Numbers) 
require("dotenv").config();
const express = require("express");
const mongoose = require("mongoose");
const session = require("express-session");
const bcrypt = require("bcryptjs");

const Admin = require("./models/Admin");
const Employee = require("./models/Employee");

const app = express();

// Middleware
app.set("view engine", "ejs");
app.use(express.urlencoded({ extended: true }));
app.use(
  session({
    secret: "my-super-secret-key",
    resave: false,
    saveUninitialized: false,
  })
);

// MongoDB Connection
mongoose
  .connect(process.env.MONGO_URI || "mongodb://localhost:27017/employee_portal", {
    useNewUrlParser: true,
    useUnifiedTopology: true,
  })
  .then(async () => {
    console.log("Connected to MongoDB");
    const adminExists = await Admin.findOne({ username: "admin" });
    if (!adminExists) {
      const hashedPassword = await bcrypt.hash("password123", 10);
      await Admin.create({ username: "admin", password: hashedPassword });
      console.log("Default admin created (Username: admin, Password: password123)");
    }
  })
  .catch((err) => console.log("MongoDB connection error:", err));

// Authentication Middleware
const requireAuth = (req, res, next) => {
  if (req.session.adminId) {
    next();
  } else {
    res.redirect("/login");
  }
};

// --- ROUTES ---
app.get("/login", (req, res) => res.render("login", { error: null }));

app.post("/login", async (req, res) => {
  const { username, password } = req.body;
  const admin = await Admin.findOne({ username });
  if (admin && (await bcrypt.compare(password, admin.password))) {
    req.session.adminId = admin._id;
    res.redirect("/");
  } else {
    res.render("login", { error: "Invalid username or password" });
  }
});

app.get("/logout", (req, res) => {
  req.session.destroy();
  res.redirect("/login");
});

// Dashboard
app.get("/", requireAuth, async (req, res) => {
  const { search, department } = req.query;
  let query = {};
  if (search) query.name = new RegExp(search, "i");
  if (department) query.department = department;

  const employees = await Employee.find(query).sort({ joiningDate: -1 });
  const departments = await Employee.distinct("department");
  res.render("dashboard", {
    employees,
    departments,
    searchQuery: search,
    deptQuery: department,
  });
});

// Add Employee
app.get("/add", requireAuth, (req, res) => res.render("add"));

app.post("/add", requireAuth, async (req, res) => {
  try {
    const newEmployee = new Employee({
      ...req.body,
      skills: req.body.skills ? req.body.skills.split(",").map((s) => s.trim()) : [],
    });
    await newEmployee.save();
    res.redirect("/");
  } catch (err) {
    res.send("Error saving employee. Ensure Emp ID is unique.");
  }
});

// View Single Employee
app.get("/view/:id", requireAuth, async (req, res) => {
  const employee = await Employee.findById(req.params.id);
  if (!employee) return res.status(404).send("Employee not found");
  res.render("view", { employee });
});

// Edit Employee
app.get("/edit/:id", requireAuth, async (req, res) => {
  const employee = await Employee.findById(req.params.id);
  res.render("edit", { employee });
});

app.post("/edit/:id", requireAuth, async (req, res) => {
  await Employee.findByIdAndUpdate(req.params.id, {
    ...req.body,
    skills: req.body.skills ? req.body.skills.split(",").map((s) => s.trim()) : [],
  });
  res.redirect("/");
});

// Delete Employee
app.post("/delete/:id", requireAuth, async (req, res) => {
  await Employee.findByIdAndDelete(req.params.id);
  res.redirect("/");
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on http://localhost:${PORT}`);
});
